## 1. 整体架构


## 2. 滚动动画系统（Reveal）

所有入场动画元素加 `class="reveal"`，CSS 初始状态：

```css
.reveal { opacity: 0; transform: translateY(18px); transition: opacity .65s ease, transform .65s ease; }
.reveal.visible { opacity: 1; transform: none; }
```

用 `IntersectionObserver` 代替 scroll 事件（性能更好，不需要 throttle）：

```js
function initReveal() {
  const io = new IntersectionObserver((entries) => {
    entries.forEach(e => {
      if (e.isIntersecting) {
        e.target.classList.add('visible');
        io.unobserve(e.target);   // 触发后不再观察，避免重复
      }
    });
  }, { threshold: 0.08, rootMargin: '0px 0px -40px 0px' });
  $$('.reveal').forEach(el => io.observe(el));
}
```

**关键**：`DOMContentLoaded` 时调一次 `initReveal()`，JS 渲染完所有 DOM 后再调一次（`setTimeout(initReveal, 100)`）。因为 JS 动态插入的 `.reveal` 元素不在第一次扫描里。

### 3.2 打分条动画

CSS transition 需要从 0 开始，不能一开始就设目标值。做法：

```js
// 先设 0
el.style.width = '0%';
// 等浏览器完成一帧再设目标值，触发 transition
setTimeout(() => {
  el.style.width = scores[idx] + '%';
  el.style.transition = 'width .9s cubic-bezier(.2,.8,.3,1)';
}, 60);
```

直接赋值 width 不经过 setTimeout/rAF 的话，浏览器会合并两次 style 修改，不会产生动画。

### 3.3 自动播放与循环

```js
function loop() {
  if (!simState.playing) return;
  simStep();
  if (simState.idx >= SIM_SCRIPT.length) {
    // 结束后等 2.4 秒自动重播
    simState.timer = setTimeout(() => {
      simReset();
      simState.playing = true;
      loop();
    }, 2400);
    return;
  }
  simState.timer = setTimeout(loop, SIM_DELAYS[simState.idx-1] || 1200);
}
```

## 4. Smart Shield · 6 维分类器（纯前端）

不调后端，前端自己算分，让评委"改一个字实时看分数变化"：

```js
function classify(text, sender, chat) {
  const has = (kw) => kw.some(k => t.includes(k));
  
  let urgent = 30 + (has(['P0','上线','紧急','立即','马上']) ? 55 : 0)
                  + (t.includes('!') || t.includes('！') ? 6 : 0);
  let self   = 30 + (has(['你','@','请','麻烦','帮我']) ? 35 : 0);
  let work   = 30 + (has(['上线','需求','PR','部署','bug','review']) ? 45 : 0);
  let time   = 30 + (has(['今晚','今天','明早','10点','DDL']) ? 50 : 0);
  
  const senderW = { boss:90, peer:60, cross:48, bot:25 }[sender] || 50;
  const chatW   = { dm:78, team:62, big:35, biz:45 }[chat] || 50;
  
  return [urgent, self, work, time, senderW, chatW].map(n => Math.max(0, Math.min(100, Math.round(n))));
}
```

加权公式（权重之和 = 1.0）：
```
total = urgent×0.22 + self×0.18 + work×0.18 + time×0.14 + sender×0.16 + channel×0.12
```

加权总分 ≥78 → P0，≥58 → P1，≥38 → P2，其余 → P3。

## 5. Mentor Skill · Typewriter 动画

每个 Skill 卡片入视口后触发 playSkill()：

```js
function playSkill(demo, script, idx) {
  demo.innerHTML = '';
  let t = 0;
  script.forEach((s, i) => {
    const delay = t;
    t += s.type === 'rewrite' ? 1100 : 500;   // 重写等待更长
    setTimeout(() => {
      const node = h('div', {class:'step'});
      if (s.type === 'rewrite') {
        const span = h('span', {class:'typed', style:{whiteSpace:'pre-line'}});
        node.appendChild(h('div',{}, [h('span',{class:'diff-add'},'＋'), ' ', span]));
        typewriter(span, s.text, 14);          // 每 14ms 打一个字
      }
      // ... 其他类型
      demo.appendChild(node);
    }, delay);
  });
}

function typewriter(el, text, ms) {
  let i = 0;
  const tick = () => {
    el.textContent = text.slice(0, i);
    if (i < text.length) { i += 1; setTimeout(tick, ms); }
    else el.style.borderRight = '0';   // 光标消失
  };
  tick();
}
```

## 6. FlowMemory 三层动画

三个节点依次高亮（每 1600ms 切换）：

```js
function playMemoryPipeline() {
  const nodes = [n1, n2, n3];
  nodes.forEach(n => n.classList.remove('active'));
  
  setTimeout(() => n1.classList.add('active'), 200);
  setTimeout(() => { n1.classList.remove('active'); n2.classList.add('active'); }, 1800);
  setTimeout(() => { n2.classList.remove('active'); n3.classList.add('active'); }, 3600);
  
  // 同时日志行逐条 append
  MEM_LINES.forEach((line, i) => {
    setTimeout(() => {
      const ln = h('div', {class:'line'}, line);
      pkg.appendChild(ln);
      if (pkg.childElementCount > 7) pkg.removeChild(pkg.firstChild); // 保持最多 7 行
    }, 400 + i * 700);
  });
}
```

6 tier markdown 每层用 CSS animation-delay 错开：

```css
.mem-tier {
  opacity: 0;
  transform: translateX(-10px);
  animation: tierFold .5s ease forwards;
}
```

第 i 层 delay = `i * 0.12s`，形成"灯笼展开"效果。

## 7. MCP Live · fetch 真实后端

```js
async function callMcp(tool) {
  const out = $('#mcpResult');
  out.textContent = `→ POST /mcp/call ...`;
  try {
    const r = await fetch('/mcp/call', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ tool: tool.name, arguments: tool.demoArgs }),
    });
    const j = await r.json();
    out.textContent = `← ${r.status} ${r.statusText}\n` + JSON.stringify(j, null, 2);
  } catch (e) {
    // graceful fallback：展示 curl 命令
    out.textContent = `× fetch 失败（本地预览/跨域）\n\ncurl 等效命令：\n  curl -XPOST http://118.178.242.26/mcp/call \\\n       -d '{"tool":"${tool.name}","arguments":...}'`;
  }
}
```

失败时不报错崩溃，而是展示等效 curl 命令，让评委知道怎么手动验证。
